639 inline bool getDateTime(m5::rtc_date_t* date, m5::rtc_time_t* time)
640 {
641 rx8130::rtc_date_t d;
642 rx8130::rtc_time_t t;
643 if (!read_datetime(date ? &d : nullptr, time ? &t : nullptr)) {
644 return false;
645 }
646 if (date) {
647 *date = {d.year, d.month, d.date, d.weekDay};

649 if (time) {
